```ts
import { dirname, join, resolve } from 'node:path';

export const GITCONFIG = join(import.meta.dir, 'tardis.gitconfig');

export type ShellResult = { output: string; code: number };
export type StageOptions = { config?: string; date?: string };

function stageEnv(cwd: string, { config = GITCONFIG, date }: StageOptions): Record<string, string> {
  const env: Record<string, string> = {
    PATH: process.env.PATH ?? '',
    HOME: process.env.HOME ?? '',
    LANG: 'C',
    // The presenter's own git config must not leak into the timeline.
    GIT_CONFIG_GLOBAL: config,
    GIT_CONFIG_NOSYSTEM: '1',
    // Before `git init`, the live dir must not resolve to an enclosing repo.
    GIT_CEILING_DIRECTORIES: dirname(resolve(cwd)),
    GIT_EDITOR: 'true',
    GIT_TERMINAL_PROMPT: '0',
    // No TTY: anything that pages would hang the request.
    PAGER: 'cat',
    GIT_PAGER: 'cat',
  };
  if (date) Object.assign(env, { GIT_AUTHOR_DATE: date, GIT_COMMITTER_DATE: date });
  return env;
}

export async function spawn(
  cmd: string[],
  cwd: string,
  options: StageOptions,
  timeoutMs: number,
  label: string,
): Promise<ShellResult> {
  const proc = Bun.spawn(cmd, { cwd, env: stageEnv(cwd, options), stdout: 'pipe', stderr: 'pipe', timeout: timeoutMs });
  const [stdout, stderr, code] = await Promise.all([
    Bun.readableStreamToText(proc.stdout),
    Bun.readableStreamToText(proc.stderr),
    proc.exited,
  ]);
  if (proc.signalCode) {
    const note = `${label} timed out after ${timeoutMs / 1000}s`;
    return { output: `${stdout}${stderr}${stdout || stderr ? '\n' : ''}${note}`, code };
  }
  return { output: stdout + stderr, code };
}

export const sh = (cwd: string, line: string, options: StageOptions = {}, timeoutMs = 10_000) =>
  spawn(['bash', '-c', line], cwd, options, timeoutMs, `\`${line}\``);
```

```ts
import type { Episode } from '../shared/types';

const FIELD = /^[A-Z][A-Za-z ]*: /;

export function applyEpisode(md: string, { set = {}, note }: Pick<Episode, 'set' | 'note'>): string {
  const lines = md.replace(/\n$/, '').split('\n');
  for (const [key, value] of Object.entries(set)) {
    const line = `${key}: ${value}`;
    const at = lines.findIndex((l) => l.startsWith(`${key}: `));
    if (at >= 0) lines[at] = line;
    else lines.splice(lines.findLastIndex((l) => FIELD.test(l)) + 1, 0, line);
  }
  if (note) lines.push(note);
  return `${lines.join('\n')}\n`;
}
```

```ts
import { join } from 'node:path';
import type { Step } from '../shared/types';
import { type ShellResult, type StageOptions, sh } from './shell';
import { applyEpisode } from './tardis';

export const TARDIS_MD = 'TARDIS.md';

export type Played = ShellResult & { cmd: string };

export async function play(dir: string, step: Step, options: StageOptions): Promise<Played> {
  if ('cmd' in step) return { cmd: step.cmd, ...(await sh(dir, step.cmd, options)) };
  const path = join(dir, TARDIS_MD);
  await Bun.write(path, applyEpisode(await Bun.file(path).text(), step.episode));
  const cmd = `git commit -am "${step.episode.title}"`;
  return { cmd, ...(await sh(dir, cmd, options)) };
}

export const meetsExpectation = (step: Step, code: number) =>
  'cmd' in step && step.fails ? code !== 0 : code === 0;
```

```ts
import { copyFile, cp, mkdir, rm } from 'node:fs/promises';
import { join } from 'node:path';
import type { EpisodeResult, LiveState, RunResult, Scene, WorkDir } from '../shared/types';
import { buildSnapshot, DAY, ROSE_AIRED } from './build';
import { readGraph } from './graph';
import { collapseCarriageReturns } from './output';
import { meetsExpectation, play } from './play';
import { GITCONFIG, sh } from './shell';
import { livePath, readWorkDir } from './workdir';

export class UnknownEpisodeError extends Error {}

/** Every repo dir carries its own global git config next to it, so aliases travel with snapshots. */
const configOf = (dir: string) => `${dir}.gitconfig`;

async function copyRepo(from: string, to: string) {
  await rm(to, { recursive: true, force: true });
  await cp(from, to, { recursive: true });
  await copyFile(configOf(from), configOf(to));
}

export class Engine {
  private readonly home: string;
  private readonly scenes: Scene[];
  private readonly live: string;
  private current: Scene | undefined;
  // enter and run share one live dir; interleaved rm/cp corrupts it.
  private queue: Promise<unknown> = Promise.resolve();

  constructor(home: string, scenes: Scene[]) {
    this.home = home;
    this.scenes = scenes;
    this.live = join(home, 'live');
  }

  private snapshot(id: string) {
    return join(this.home, 'snapshots', id);
  }

  async prepare() {
    const later = this.scenes.slice(1).find((s) => s.build);
    if (later) throw new Error(`only the first scene has a build: ${later.id}`);

    const work = join(this.home, 'work');
    const build = this.scenes[0]?.build ?? [];
    await buildSnapshot(work, build);
    await copyFile(GITCONFIG, configOf(work));
    await mkdir(join(this.home, 'snapshots'), { recursive: true });

    // Replays continue the build's fixed clock, so snapshot hashes are stable.
    let tick = build.length;
    for (const scene of this.scenes) {
      await copyRepo(work, this.snapshot(scene.id));
      for (const [i, step] of scene.steps.entries()) {
        const date = `@${ROSE_AIRED + tick++ * DAY} +0000`;
        const played = await play(work, step, { config: configOf(work), date });
        if (!meetsExpectation(step, played.code))
          throw new Error(`${scene.id} step ${i + 1} (${played.cmd}) exited ${played.code}: ${played.output}`);
      }

  private serialise<T>(task: () => Promise<T>): Promise<T> {
    const result = this.queue.then(task);
    this.queue = result.catch(() => undefined);
    return result;
  }

  enter(id: string): Promise<LiveState> {
    return this.serialise(() => this.doEnter(id));
  }

  run(input: string): Promise<RunResult> {
    return this.serialise(async () => this.withLive(await sh(this.live, input, { config: configOf(this.live) })));
  }

  episode(code: string): Promise<EpisodeResult> {
    return this.serialise(async () => {
      const step = this.current?.steps.find((s) => 'episode' in s && s.episode.code === code);
      if (!step) throw new UnknownEpisodeError(`unknown episode: ${code}`);
      const { cmd, ...result } = await play(this.live, step, { config: configOf(this.live) });
      return { cmd, ...(await this.withLive(result)) };
    });
  }

  readFile(path: string): Promise<string> {
    return this.serialise(async () => Bun.file(await livePath(this.live, path)).text());
  }

  writeFile(path: string, content: string): Promise<WorkDir> {
    return this.serialise(async () => {
      await Bun.write(await livePath(this.live, path), content);
      return readWorkDir(this.live);
    });
  }

  private async withLive(result: { output: string; code: number }): Promise<RunResult> {
    return {
      ...result,
      output: collapseCarriageReturns(result.output),
      graph: await readGraph(this.live),
      workdir: await readWorkDir(this.live),
    };
  }

  private async doEnter(id: string): Promise<LiveState> {
    const scene = this.scenes.find((s) => s.id === id);
    if (!scene) throw new Error(`unknown scene: ${id}`);
    await copyRepo(this.snapshot(id), this.live);
    this.current = scene;
    return { graph: await readGraph(this.live), workdir: await readWorkDir(this.live) };
  }
}
```

```tsx
import { useCallback, useEffect, useRef, useState } from 'react';
import type { Episode } from '../shared/types';

type Props = { episode: Episode; onPlay: () => void; seconds?: number };

export function NextEpisode({ episode, onPlay, seconds = 8 }: Props) {
  const [left, setLeft] = useState(seconds);
  const played = useRef(false);
  const playOnce = useCallback(() => {
    if (played.current) return;
    played.current = true;
    onPlay();
  }, [onPlay]);

  useEffect(() => {
    if (left === 0) {
      playOnce();
      return;
    }
    const timer = setTimeout(() => setLeft((s) => s - 1), 1000);
    return () => clearTimeout(timer);
  }, [left, playOnce]);

  useEffect(() => {
    const onKey = (e: KeyboardEvent) => {
      if (e.key !== 'n' || e.target instanceof HTMLTextAreaElement) return;
      if (e.target instanceof HTMLInputElement && e.target.value !== '') return;
      e.preventDefault();
      playOnce();
    };
    window.addEventListener('keydown', onKey);
    return () => window.removeEventListener('keydown', onKey);
  }, [playOnce]);

  return (
    <aside className="next-episode" aria-label="next episode">
      <small>NEXT EPISODE</small>
      <div className="code">{episode.code}</div>
      <div className="title">{episode.title}</div>
      <button type="button" onClick={playOnce}>
        ▶ Play <span className="countdown">{left}</span>
      </button>
```
